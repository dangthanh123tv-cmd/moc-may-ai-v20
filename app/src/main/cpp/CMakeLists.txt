cmake_minimum_required(VERSION 3.22.1)
project(mocmay_llm LANGUAGES C CXX)

option(MOCMAY_WITH_LLAMA_CPP "Build against vendored llama.cpp" OFF)
set(MOCMAY_LLAMA_CPP_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp" CACHE PATH "llama.cpp source directory")

add_library(mocmay_llm SHARED mocmay_llm_jni.cpp)
find_library(log-lib log)
target_link_libraries(mocmay_llm PRIVATE ${log-lib})
target_compile_features(mocmay_llm PRIVATE cxx_std_17)

if(MOCMAY_WITH_LLAMA_CPP)
    if(NOT EXISTS "${MOCMAY_LLAMA_CPP_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "MOCMAY_WITH_LLAMA_CPP=ON but llama.cpp was not found at ${MOCMAY_LLAMA_CPP_DIR}. Run scripts/fetch-llama.sh first.")
    endif()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    add_subdirectory("${MOCMAY_LLAMA_CPP_DIR}" "${CMAKE_BINARY_DIR}/llama.cpp")
    target_compile_definitions(mocmay_llm PRIVATE MOCMAY_HAS_LLAMA=1)
    target_include_directories(mocmay_llm PRIVATE "${MOCMAY_LLAMA_CPP_DIR}/include")
    target_link_libraries(mocmay_llm PRIVATE llama)
else()
    target_compile_definitions(mocmay_llm PRIVATE MOCMAY_HAS_LLAMA=0)
endif()
